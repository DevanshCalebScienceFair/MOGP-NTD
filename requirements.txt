# Versions pinned to the mogp-drug env that this checkout is known to run on
# (Python 3.11). The loose bounds this file used to carry resolved to a stack
# that does not work here, so they are now exact.
#
#   conda env: python 3.11 + vina + openbabel + ninja + rdkit  (conda-forge)
#   everything below: pip
#
# >>> DO NOT `pip install -r requirements.txt` INTO A FRESH ENV. <<<
# It will fail with ResolutionImpossible. PyTDC 1.1.15 declares numpy<2.0 while
# the rest of the stack needs numpy 2.x, so this set cannot be produced in one
# resolver pass — it has to be installed in phases. Use the script instead:
#
#   bash setup_env.sh [env-name]
#
# This file remains the authoritative record of WHICH versions the env carries,
# and `pip install --dry-run -r requirements.txt` against the built env is a
# useful drift check. It is just not a one-shot installer.
#
# NOTE: rdkit is installed from CONDA, not pip, and is therefore absent from
# this file on purpose. The pip wheel for cp311 is compiled against NumPy 1.x
# and dies with "AttributeError: _ARRAY_API not found" under NumPy 2. Installing
# it here would silently break the env. Get it with:
#   conda install -c conda-forge rdkit
PyTDC==1.1.15
# Pinned: the serialized models in models/pretrained_admet/ were saved with
# scikit-learn 1.9.0. Loading them under a different version triggers
# InconsistentVersionWarning and may silently produce invalid predictions.
# 1.9.0 requires Python >= 3.11 — this is why the env is 3.11 and not 3.10.
scikit-learn==1.9.0
joblib==1.5.3
pandas==2.3.3
numpy==2.2.6
botorch==0.18.1
# Paired significance tests (Wilcoxon signed-rank + paired t-test) in
# run_benchmark_seeds.py. Usually already present as a botorch/sklearn dep.
scipy==1.17.1
pytest==9.1.1
# dashboard.py (results viewer) only:
streamlit==1.61.1
matplotlib==3.11.1

# --- docking stack -----------------------------------------------------------
# meeko prepares the ligand PDBQT; biopython parses the receptor PDB.
meeko==0.7.1
biopython==1.88
# meeko 0.7.1 imports gemmi at package import time but does not declare it, so
# `import docking` fails with ModuleNotFoundError without this line.
gemmi==0.7.5

# --- transitive pins that are load-bearing -----------------------------------
# PyTDC pins pyarrow==12.0.1, which is compiled against NumPy 1.x and raises
# "_ARRAY_API not found" the moment pandas imports it under NumPy 2. 24.0.0 is
# the version the working env actually carries.
pyarrow==24.0.0
# PyTDC imports pkg_resources, which setuptools REMOVED in 82.0. Anything >=82
# breaks `import data` with ModuleNotFoundError: No module named 'pkg_resources'.
setuptools==81.0.0
